function Test-Endpoint {
    param(
        [string]$Name,
        [string]$Url,
        [hashtable]$Body,
        [string]$Method = "POST",
        [bool]$ExpectError = $false
    )
    
    Write-Host "Testing: $Name..." -ForegroundColor Yellow -NoNewline
    
    try {
        if($Method -eq "GET") {
            $response = Invoke-RestMethod -Uri $Url -Method GET -TimeoutSec 10
        } else {
            $json = $Body | ConvertTo-Json
            $response = Invoke-RestMethod -Uri $Url -Method POST -Body $json -ContentType "application/json" -TimeoutSec 10
        }
        
        if($ExpectError) {
            Write-Host " ⚠️ UNEXPECTED SUCCESS (expected error)" -ForegroundColor Yellow
            $script:testsSkipped++
            return $false
        }
        
        # Check for success indicators
        $isSuccess = $false
        if($response.status -eq "success") { $isSuccess = $true }
        if($response.status -eq "healthy") { $isSuccess = $true }
        if($response.status -eq "running") { $isSuccess = $true }
        if($response.status -eq "Trading Webhook Active") { $isSuccess = $true }
        if($response.status -eq "online") { $isSuccess = $true }
        
        if($isSuccess) {
            Write-Host " ✅ PASS" -ForegroundColor Green
            $script:testsPassed++
            return $true
        } else {
            Write-Host " ⚠️ UNKNOWN RESPONSE" -ForegroundColor Yellow
            $script:testsSkipped++
            return $false
        }
    } catch {
        if($ExpectError) {
            Write-Host " ✅ PASS (expected error)" -ForegroundColor Green
            $script:testsPassed++
            return $true
        }
        
        Write-Host " ❌ FAIL" -ForegroundColor Red
        Write-Host "  Error: $($_.Exception.Message)" -ForegroundColor Red
        $script:testsFailed++
        return $false
    }
}
